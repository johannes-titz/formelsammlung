\subsection*{Kovarianz}
$$s_{x, y}=\frac{1}{n}\sum\limits_{i=1}^{n}(x_i-\bar{x})(y_i-\bar{y})$$

\subsection*{Korrelation}
Pearson-Korrelationskoeffizient \\
(Produkt-Moment-Korrelation)
$$r_{x, y}=\frac{s_{x, y}}{s_xs_y}= \frac{1}{n}\sum\limits_{i=1}^{n}{z_{x_i}\cdot z_{y_i}}$$

\subsection*{Partialkorrelation}
$$r_{x, y.z}=\frac{r_{x, y}- r_{x, z}\cdot r_{y, z}}{\sqrt{1-r_{x, z}^{2}}\cdot\sqrt{1-r_{y, z}^{2}}}$$

\subsection*{Phi-Koeffizient (für dichotome Variablen)}
$$\varphi=\frac{a\cdot d-b\cdot c}{\sqrt{(a+b)(c+d)(a+c)(b+d)}}$$

\begin{center}
\begin{tabular}{@{} l c c c @{}}

\subsection*{Yules Q (für dichotome Variablen)}

$$Q = \frac{ad-bc}{ad+bc}$$

\subsection*{Interrater-Reliabilität (Kappa)}

\begin{align*}
  p_o &= \frac{1}{n}\sum_{i=1}^k f_{ii}\\
  p_e &= \sum_{i=1}^k p_{i\cdot}p_{\cdot i}
  = \frac{1}{n^2}\sum_{i=1}^k f_{i\cdot}f_{\cdot i}\\
  \kappa &= \frac{p_o - p_e}{1 - p_e}
\end{align*}

\begin{tabular}{rl}
  $p_o$:&beobachtete Übereinstimmung\\
  $p_e$:&erwartete Übereinstimmung unter Zufall\\
  $p_{i\cdot}$:&die Randwahrscheinlichkeit der Zeile $i$\\
  $p_{\cdot i}$:&die Randwahrscheinlichkeit der Spalte $i$\\
  $k$:&die Anzahl der Kategorien
\end{tabular}

\par\smallskip
Die Randhäufigkeiten $f_{i\cdot}$ und $f_{\cdot i}$ sind die Zeilen- und Spaltensummen. Division durch $n$ ergibt die Randwahrscheinlichkeiten.
