%\section{Konfidenzintervalle}

\begin{multicols}{2}

Für zweiseitige $(1-\alpha)$-Konfidenzintervalle:

\subsection*{Binomialverteilung (Approximation)}
Die folgende Approximation über die Normalverteilung ist nur nutzbar, wenn $n \cdot \hat{\pi} \cdot (1-\hat{\pi})>9$ \\\\

für Anteile
$$\hat{\pi} \pm \hat{\sigma}_{\mathrm{Anteil}}\cdot z_{1-\alpha/2}$$

für Anzahlen/Objekte
$$\hat{\pi} \cdot n \pm \hat{\sigma}_{\mathrm{Anzahl}}\cdot z_{1-\alpha/2}$$

\subsection*{Mittelwert} 
$$\bar{x} \pm \hat{\sigma}_{\bar{x}} \cdot t_{\mathrm{df},\,1-\alpha/2}$$

\subsection*{MWU, US} 
$$(\bar{x}_\mathrm{A}-\bar{x}_\mathrm{B}) \pm \hat{\sigma}_{\bar{x}_\mathrm{A}-\bar{x}_\mathrm{B}}\cdot t_{\mathrm{df},\,1-\alpha/2}$$

\subsection*{MWU, AS}
$${\bar{D}} \pm \hat{\sigma}_{\bar{D}}\cdot t_{\mathrm{df},\,1-\alpha/2}$$

\subsection*{Einfache lineare Regression}
% Konfidenzintervall fuer den Anstieg
\[
b_1
\pm
t_{n-2,\,1-\alpha/2}
\cdot\hat{\sigma}_{b_1}
\]

% Konfidenzintervall fuer die Konstante
\[
b_0
\pm
t_{n-2,\,1-\alpha/2}
\cdot \hat{\sigma}_{b_0}
\]

\end{multicols}


\end{formulaBox}
