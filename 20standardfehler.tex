\begin{formulaBox}{Standardfehler für Stichprobenverteilungen}{ML I}
\begin{multicols}{2}
\subsection*{Binomialverteilung}

$$\sigma_{\mathrm{Anzahl}}=\sqrt{n \pi (1-\pi)}$$

$$\hat{\sigma}_{\mathrm{Anzahl}}=\sqrt{n \hat{\pi} (1-\hat{\pi})}$$

$$\sigma_{\mathrm{Anteil}}=\sqrt{\frac{\pi (1-\pi)}{n}}$$

$$\hat{\sigma}_{\mathrm{Anteil}}=\sqrt{\frac{\hat{\pi} (1-\hat{\pi})}{n}}$$

\subsection*{Mittelwerte}

$$\hat{\sigma}_{\bar{x}}=\frac{\hat{\sigma}}{\sqrt{n}}=\frac{s}{\sqrt{n-1}}$$

\subsection*{Einfache lineare Regression}

\[
\hat{\sigma}_{b_1}
=
\sqrt{
\frac{\hat{\sigma}^2}
{\sum_{i=1}^n (x_i - \bar x)^2}
}
\]

\[
  \hat{\sigma}_{b_0}
=
\sqrt{
\hat{\sigma}^2
\left(
\frac{1}{n}
+
\frac{\bar x^2}
{\sum_{i=1}^n (x_i - \bar x)^2}
\right)
}
\]

\columnbreak

\subsection*{MWU für US}

Bei unabhängigen Stichproben:

\[
\hat{\sigma}_{\bar{x}_\mathrm{A}-\bar{x}_\mathrm{B}}
=\sqrt{\frac{\hat{\sigma}_\mathrm{A}^2}{n_\mathrm{A}}
+\frac{\hat{\sigma}_\mathrm{B}^2}{n_\mathrm{B}}}
\]

Bei gleichen Varianzen in den beiden Populationen kann die Varianz gepoolt werden:

$$\hat{\sigma}_{\bar{x}_\mathrm{A}-\bar{x}_\mathrm{B}}=\sqrt{\frac{(n_\mathrm{A}-1)\cdot\hat{\sigma}^2_\mathrm{A}+(n_\mathrm{B}-1)\cdot\hat{\sigma}^2_\mathrm{B}}{(n_\mathrm{A}-1)+(n_\mathrm{B}-1)}(\frac{1}{n_\mathrm{A}}+\frac{1}{n_\mathrm{B}})}$$

Für $n_\mathrm{A}=n_\mathrm{B}=n$ vereinfacht sich die Formel zu:

\[
\hat{\sigma}_{\bar{x}_\mathrm{A} - \bar{x}_\mathrm{B}}
= \sqrt{\frac{\hat{\sigma}_\mathrm{A}^2}{n}
+ \frac{\hat{\sigma}_\mathrm{B}^2}{n}}
\]

\subsection*{MWU für AS}

$$\hat{\sigma}_{\bar{D}}=\frac{\hat{\sigma}_D}{\sqrt{n}}$$
