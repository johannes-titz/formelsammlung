\begin{multicols}{2}
\subsection*{Gewichteter Mittelwert}
an der Stichprobengröße $N$ gewichtet (Vereinfachung, nicht die Standardmethode):
$$\bar{\theta}_N = \frac{\sum_{i} N_i \theta_i}{\sum_{i} N_i}$$
Inverse-Varianz-Gewichtung:
$$\bar{\theta}_{\mathrm{IV}} = \frac{\sum_i W_i \theta_i}{\sum_i W_i}, \quad W_i = \frac{1}{V_i}$$
$$\mathrm{SE}(\bar{\theta}) = \sqrt{\frac{1}{\sum_i W_i}}$$

% see Borenstein equation 4.20; we probably can get it a bit more precise it appears
% Borenstein 4.23-4.25
\subsection*{Varianz von $d$ und $g$}
$$\mathrm{Var}(d) \approx \frac{n_A+n_B}{n_A\, n_B} + \frac{d^2}{2\,(n_A+n_B)}$$
$$\mathrm{Var}(g) = J^2\,\mathrm{Var}(d), \quad \mathrm{SE}(g) = J\cdot\mathrm{SE}(d)$$
mit $\mathrm{SE}=\sqrt{\mathrm{Var}}$ und $J$ (Bias-Korrektur, siehe Effektgrößen).

% Borenstein 6.2-6.4
\subsection*{Fisher-$z$ (für Korrelationen)}
$$z = \operatorname{artanh}(r) = \tfrac{1}{2}\ln\frac{1+r}{1-r}, \quad r = \tanh(z)$$
$$V_z \approx \frac{1}{n-3}, \quad W_i = n_i - 3$$

% df = Q unter H0!
% Borenstein 16.4, 16.9
\subsection*{Modell und Heterogenität}
Fixed-Effects: $W_i = \frac{1}{V_i}$; \ Random-Effects: $W_i^{*} = \frac{1}{V_i+\tau^2}$

$$Q = \sum_i W_i\,(\theta_i - \bar{\theta})^2, \quad Q_{H_0} = \mathrm{df}=k-1$$

$$I^2 = \frac{Q-\mathrm{df}}{Q}\cdot 100\,\%$$
Hinweis: Unter der $H_0$ folgt $Q$ einer $\chi^2$-Verteilung, sodass der Erwartungswert den Freiheitsgraden entspricht.
\end{multicols}
\end{formulaBox}
